# Canonical forms: one function written with its operands in other orders or groupings, or its
# constants in other places. A tool that normalizes sees these by comparing forms. Format:
# bitvector.txt.

## Commutativity and associativity
add-reordered [bool]: (= (bvadd x (bvadd y z)) (bvadd (bvadd z y) x)) ==> true
mul-reordered [bool]: (= (bvmul x (bvmul y z)) (bvmul (bvmul z x) y)) ==> true
and-reordered [bool]: (= (bvand x (bvand y z)) (bvand (bvand z y) x)) ==> true
or-reordered [bool]: (= (bvor x (bvor y z)) (bvor (bvor y z) x)) ==> true
xor-reordered [bool]: (= (bvxor x (bvxor y z)) (bvxor (bvxor z x) y)) ==> true
add-sub-reordered [bool]: (= (bvsub (bvadd x y) z) (bvadd (bvsub x z) y)) ==> true
sub-sub-reordered [bool]: (= (bvsub (bvsub x y) z) (bvsub (bvsub x z) y)) ==> true
eq-symmetric [bool]: (= (= x y) (= y x)) ==> true
distinct-symmetric [bool]: (= (distinct x y) (distinct y x)) ==> true
sum-minus-reordered-sum: (bvsub (bvadd (bvadd x y) z) (bvadd z (bvadd x y))) ==> [0]
product-minus-reordered-product: (bvsub (bvmul (bvmul x y) z) (bvmul z (bvmul y x))) ==> [0]
xor-with-reordered-xor: (bvxor (bvxor x (bvxor y z)) (bvxor (bvxor z y) x)) ==> [0]
and-with-reordered-and: (bvand (bvand x y) (bvand y x)) ==> (bvand x y)
or-with-reordered-or: (bvor (bvor x y) (bvor y x)) ==> (bvor x y)
sum-of-swapped-products: (bvsub (bvadd (bvmul x y) (bvmul z x)) (bvadd (bvmul x z) (bvmul y x))) ==> [0]
